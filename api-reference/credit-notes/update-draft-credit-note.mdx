---
title: "Update draft credit note"
description: "Modifies the details of an existing unissued draft credit note. Once a credit note is ISSUED, it can no longer be edited via this endpoint."
api: "PUT /credit-notes/{id}/draft"
---

### Headers

<ParamField header="X-API-Version" type="string" default="1.2" required>
  The API version to use. Supported versions are `1.1` and `1.2`.
</ParamField>

<ParamField header="Idempotency-Key" type="string">
  A unique key so a retried request is not processed twice.
</ParamField>

### Path Parameters

<ParamField path="id" type="string" required>
  The unique ID (UUID) of the draft credit note to update.
</ParamField>

### Request Body Parameters

<ParamField body="amount" type="number" required>
  Updated credit amount. Must be greater than 0.
</ParamField>

<ParamField body="currency" type="string">
  Currency code (e.g. `AUD`).
</ParamField>

<ParamField body="reason" type="string" required>
  Updated reason for issuing the credit.
</ParamField>

<ParamField body="description" type="string">
  Updated memo or internal accounting remarks.
</ParamField>

<ParamField body="invoice_id" type="string">
  Updated target invoice UUID, if reassigning to a different invoice.
</ParamField>

### Response Body

<ResponseField name="status" type="number">
  HTTP status code (200).
</ResponseField>

<ResponseField name="message" type="string">
  Confirmation message describing the update.
</ResponseField>

<ResponseField name="data" type="object">
  The updated draft credit note entity.
</ResponseField>

### Response Codes

| Code | Description |
|---|---|
| 200 | Draft credit note updated successfully. |
| 400 | Bad Request. Validation failure or credit note is already issued. |
| 401 | Unauthorized. Missing or invalid credentials. |
| 404 | Not Found. Draft credit note not found. |
| 500 | Internal server error. |

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.Base64;

public class Example {
    public static void main(String[] args) throws Exception {
        String credentials = Base64.getEncoder()
            .encodeToString("YOUR_UUID:YOUR_SECRET".getBytes());

        String body = String.join("\n",
            "{",
            "  \"amount\": 150,",
            "  \"currency\": \"AUD\",",
            "  \"reason\": \"Customer discount / Service dispute\",",
            "  \"description\": \"150 AUD credit issued for overbilled service hours\"",
            "}"
        );

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-payments.mygigsters.com.au/api/v1/credit-notes/3/draft"))
            .header("Authorization", "Basic " + credentials)
            .header("idempotency-Key", "cn-req-edit-002")
            .header("X-API-Version", "1.2")
            .header("Content-Type", "application/json")
            .method("PUT", HttpRequest.BodyPublishers.ofString(body))
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
```
