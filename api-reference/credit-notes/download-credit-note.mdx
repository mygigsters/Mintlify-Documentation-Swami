---
title: "Download credit note PDF"
description: "Generates and streams the formal credit note PDF document containing company details, credited items, tax information, and issued status."
api: "GET /credit-notes/{id}/download"
---

### Headers

<ParamField header="X-API-Version" type="string" default="1.2" required>
  The API version to use. Supported versions are `1.1` and `1.2`.
</ParamField>

### Path Parameters

<ParamField path="id" type="string" required>
  The unique ID (UUID) of the credit note to download.
</ParamField>

### Response Body

| Type | Description |
|---|---|
| application/pdf | Binary PDF stream of the branded credit note document. |

### Response Codes

| Code | Description |
|---|---|
| 200 | PDF generated and returned as binary document. |
| 401 | Unauthorized. Missing or invalid credentials. |
| 404 | Not Found. Credit note not found. |
| 500 | Internal server error. |

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.Base64;

public class Example {
    public static void main(String[] args) throws Exception {
        String credentials = Base64.getEncoder()
            .encodeToString("YOUR_UUID:YOUR_SECRET".getBytes());

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-payments.mygigsters.com.au/api/v1/credit-notes/cn_df8913b4-16a7-47ab-b295-ffc129e01861/download"))
            .header("Authorization", "Basic " + credentials)
            .header("X-API-Version", "1.2")
            .header("Content-Type", "application/json")
            .method("GET", HttpRequest.BodyPublishers.noBody())
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
```
