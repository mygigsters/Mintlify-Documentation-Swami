---
title: "Retrieve worker"
description: "This endpoint allows you to get details of a worker using their unique worker ID."
api: "GET /worker/{worker_id}/details"
---

### Headers

<ParamField header="X-API-Version" type="string" default="1.2" required>
  The API version to use. Supported versions are `1.1` and `1.2`.
</ParamField>

### Path Parameters

<ParamField path="worker_id" type="string" required>
  The unique identifier of the worker.
</ParamField>

### Response Body

| Field | Type | Description |
|---|---|---|
| `status` | number | Indicates the status code of the response. |
| `message` | string | A message related to the worker details request. |
| `worker`object | object | The worker object containing detailed information. |
| ↳`id` | string | Unique identifier for the worker. |
| ↳`full_name` | string | The full name of the worker. |
| ↳`email` | string | The email address of the worker. |
| ↳`phone` | string | The phone number of the worker. |
| ↳`abn` | string | The Australian Business Number of the worker. |
| ↳`bsb` | string | The BSB number of the worker (Australian bank accounts). |
| ↳`account_number` | string | The bank account number of the worker. |
| ↳`account_name` | string | The name associated with the bank account of the worker. |
| ↳`is_early_pay_enabled` | boolean | Indicates whether early pay is enabled for the worker. |
| ↳`early_pay_rates` | number | The early pay rate/percentage for the worker. Returned only when early pay is enabled. |

### Response Codes

| Code | Description |
|---|---|
| 200 | If the request is successful, the details of the worker are returned. |
| 400 | Bad Request. Invalid format or parameters. |
| 401 | Unauthorized. Missing or invalid credentials. |
| 404 | Not Found. The specified workerId does not exist. |
| 500 | Internal Server Error. |

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.Base64;

public class Example {
    public static void main(String[] args) throws Exception {
        String credentials = Base64.getEncoder()
            .encodeToString("YOUR_UUID:YOUR_SECRET".getBytes());

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-payments.mygigsters.com.au/api/v1/worker/afd629c6-8e3d-4701-b168-dec8608130a5/details"))
            .header("Authorization", "Basic " + credentials)
            .header("X-API-Version", "1.2")
            .header("Content-Type", "application/json")
            .method("GET", HttpRequest.BodyPublishers.noBody())
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
