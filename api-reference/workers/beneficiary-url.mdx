---
title: "Beneficiary URL"
description: "This endpoint allows you to generate a unique onboarding/form URL for a beneficiary/worker."
api: "POST /beneficiary-url"
---

### Headers

<ParamField header="X-API-Version" type="string" default="1.2" required>
  The API version to use. Supported versions are `1.1` and `1.2`.
</ParamField>

### Request Body Parameters

<ParamField body="worker_id" type="string" required>
  The unique identifier of the worker.
</ParamField>

<ParamField body="return_url" type="string" required>
  The URL to which the user will be redirected after completing the form.
</ParamField>

### Response Body

<ResponseField name="status" type="number">
  Indicates the status code of the request.
</ResponseField>

<ResponseField name="message" type="string">
  A message related to the status of the request.
</ResponseField>

<ResponseField name="beneficiaryUrl" type="string">
  The URL to the beneficiary form.
</ResponseField>

### Response Codes

| Code | Description |
|---|---|
| 200 | If the request is successful, the beneficiary URL is generated and returned. |
| 400 | Bad Request. Invalid request payload or missing parameters. |
| 401 | Unauthorized. Missing or invalid credentials. |
| 404 | Not Found. The specified workerId does not exist. |
| 500 | Internal Server Error. |

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.Base64;

public class Example {
    public static void main(String[] args) throws Exception {
        String credentials = Base64.getEncoder()
            .encodeToString("YOUR_UUID:YOUR_SECRET".getBytes());

        String body = String.join("\n",
            "{",
            "  \"worker_id\": \"afd629c6-8e3d-4701-b168-dec8608130a5\",",
            "  \"return_url\": \"https://example.com\"",
            "}"
        );

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-payments.mygigsters.com.au/api/v1/beneficiary-url"))
            .header("Authorization", "Basic " + credentials)
            .header("X-API-Version", "1.2")
            .header("Content-Type", "application/json")
            .method("POST", HttpRequest.BodyPublishers.ofString(body))
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
```
