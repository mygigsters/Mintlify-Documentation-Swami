---
title: "List API keys"
description: "This endpoint lists API keys. The API Key is to be used in Basic Authentication format eg. curl -u 'apiUsername:apiPassword'. To access this endpoint you need to obtain a JWT via Get OTP and Verify OTP sections."
api: "GET https://qa-super.mygigsters.com.au/auth/list"
authMethod: "bearer"
---

### Responses

| Code | Description |
|---|---|
| `200` | The API Key objects |

### Response fields

<ResponseField name="id" type="string">
  The API Key ID.
</ResponseField>

<ResponseField name="name" type="string">
  Friendly name of the API Key
</ResponseField>

<ResponseField name="apiUsername" type="string">
  The Basic Auth API Key user name
</ResponseField>

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class Example {
    public static void main(String[] args) throws Exception {
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-super.mygigsters.com.au/auth/list"))
            .header("Authorization", "Bearer JWT_FROM_VERIFY_OTP_STEP")
            .method("GET", HttpRequest.BodyPublishers.noBody())
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
```

```go Go
package main

import (
	"fmt"
	"io"
	"net/http"
)

func main() {
	req, err := http.NewRequest("GET", "https://qa-super.mygigsters.com.au/auth/list", nil)
	if err != nil {
		panic(err)
	}
	req.Header.Set("Authorization", "Bearer JWT_FROM_VERIFY_OTP_STEP")

	res, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, _ := io.ReadAll(res.Body)
	fmt.Println(string(data))
}
```
