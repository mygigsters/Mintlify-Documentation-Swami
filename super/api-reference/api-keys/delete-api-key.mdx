### Request Body

<ParamField body="id" type="string" required>
  The ID of your key (not the friendly name)
</ParamField>

### Responses

| Code | Description |
|---|---|
| `200` | API Key deleted |
| `400` | API Key not provided |
| `404` | API Key not found |

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class Example {
    public static void main(String[] args) throws Exception {
        String body = String.join("\n",
            "{",
            "  \"id\": \"1231412313213\"",
            "}"
        );

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-super.mygigsters.com.au/auth/delete-key"))
            .header("Authorization", "Bearer JWT_FROM_VERIFY_OTP_STEP")
            .header("Content-Type", "application/json")
            .method("DELETE", HttpRequest.BodyPublishers.ofString(body))
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
```

```go Go
package main

import (
	"fmt"
	"io"
	"net/http"
	"strings"
)

func main() {
	body := strings.NewReader(`{
  "id": "1231412313213"
}`)

	req, err := http.NewRequest("DELETE", "https://qa-super.mygigsters.com.au/auth/delete-key", body)
	if err != nil {
		panic(err)
	}
	req.Header.Set("Authorization", "Bearer JWT_FROM_VERIFY_OTP_STEP")
	req.Header.Set("Content-Type", "application/json")

	res, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, _ := io.ReadAll(res.Body)
	fmt.Println(string(data))
}
```
