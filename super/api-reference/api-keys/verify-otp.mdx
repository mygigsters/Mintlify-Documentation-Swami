---
title: "Verify OTP"
description: "This route returns the JWT for use with API Keys routes (/api-keys/) - note that the OTP is always 123456 for non-production environments"
api: "POST https://qa-super.mygigsters.com.au/auth/verify-otp"
authMethod: "none"
---

### Request Body

<ParamField body="otp" type="string" required>
  One Time Pin to be sent in string format
</ParamField>

<ParamField body="email" type="string" required>
  The email address which you requested the OTP for
</ParamField>

### Responses

| Code | Description |
|---|---|
| `200` | Response with the JWT to use for the rest of the API Keys routes |
| `400` | Missing info in request body |
| `404` | Cannot find user or invalid OTP |

### Response fields

<ResponseField name="message" type="string">
  Information about response
</ResponseField>

<ResponseField name="clientId" type="integer">
  Client ID of the user
</ResponseField>

<ResponseField name="email" type="string">
  Email of user
</ResponseField>

<ResponseField name="token" type="string">
  JWT for use with API Keys routes
</ResponseField>

<Accordion title="Error response fields">

<ResponseField name="errors" type="string">
  Information about missing data
</ResponseField>

</Accordion>

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class Example {
    public static void main(String[] args) throws Exception {
        String body = String.join("\n",
            "{",
            "  \"otp\": \"123456\",",
            "  \"email\": \"email@email.com\"",
            "}"
        );

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-super.mygigsters.com.au/auth/verify-otp"))
            .header("Content-Type", "application/json")
            .method("POST", HttpRequest.BodyPublishers.ofString(body))
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
```

```go Go
package main

import (
	"fmt"
	"io"
	"net/http"
	"strings"
)

func main() {
	body := strings.NewReader(`{
  "otp": "123456",
  "email": "email@email.com"
}`)

	req, err := http.NewRequest("POST", "https://qa-super.mygigsters.com.au/auth/verify-otp", body)
	if err != nil {
		panic(err)
	}
	req.Header.Set("Content-Type", "application/json")

	res, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, _ := io.ReadAll(res.Body)
	fmt.Println(string(data))
}
```
