---
title: "Remove webhook"
description: "Removes the registered webhook URL for your client. After deletion, no further events will be delivered. The webhook secret is also invalidated."
api: "DELETE https://qa-super.mygigsters.com.au/v1/webhooks"
---

To register a new webhook URL after removal, call `POST /v1/webhooks` again.

### Responses

| Code | Description |
|---|---|
| `200` | Webhook removed successfully |
| `401` | Unauthorized — invalid authentication credentials |
| `404` | No webhook registered for this client |

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.Base64;

public class Example {
    public static void main(String[] args) throws Exception {
        String credentials = Base64.getEncoder()
            .encodeToString("apiUsername:apiPassword".getBytes());

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://qa-super.mygigsters.com.au/v1/webhooks"))
            .header("Authorization", "Basic " + credentials)
            .method("DELETE", HttpRequest.BodyPublishers.noBody())
            .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
            .send(request, HttpResponse.BodyHandlers.ofString());

        System.out.println(response.body());
    }
}
```

```go Go
	}

	res, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, _ := io.ReadAll(res.Body)
	fmt.Println(string(data))
}
```
